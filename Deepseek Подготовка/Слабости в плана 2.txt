#!/usr/bin/env python3
"""
plan_worker.py — разширение на pipeline.py за новия 16-седмичен план.
Работи УСПОРЕДНО с pipeline.py (не го пипа).

Употреба:
    python plan_worker.py                    # 24/7 за всички micro_topics
    python plan_worker.py --once             # 1 batch

    python plan_worker.py --stats

import time
from pathlib import Path
from dataclasses import dataclass, field
from typing import Optional
from concurrent.futures import ThreadPoolExecutor, as_completed

# Reuse всичко от pipeline.py
from pipeline import (
    AIClient, CONFIG, State, log,

BASE_DIR = Path(__file__).parent.resolve()
PLAN_CONFIG = BASE_DIR / "plan_config.json"
PLAN_OUTPUT = BASE_DIR / "plan_output"
PLAN_OUTPUT.mkdir(exist_ok=True)

# ── Нови prompts ──

# DATA MODEL

@dataclass
class MicroTopic:
    id: str
    title: str
    problem_num: int
    difficulty: int
    estimated_minutes: int
    source_problems: list
    key_techniques: list
    misconceptions: list
    generate: dict
    
    # Status
    status: str = "pending"  # pending / article_done / problems_done / practice_done / done / error
    attempts: int = 0
    error: str = ""
    
    # Results
    article: dict = field(default_factory=dict)
    problems: dict = field(default_factory=dict)
    practice: dict = field(default_factory=dict)
    
    # Meta
    week: int = 0
    phase: int = 0


# ═══════════════════════════════════════════════════════════
# PLAN LOADER
# ═══════════════════════════════════════════════════════════
def load_plan() -> tuple[list[MicroTopic], dict]:
    """Чете plan_config.json → списък от MicroTopic."""
    if not PLAN_CONFIG.exists():
        raise FileNotFoundError(f"Липсва {PLAN_CONFIG}. Създай го първо.")
    
    with open(PLAN_CONFIG, "r", encoding="utf-8") as f:
        cfg = json.load(f)
    
    topics = []
    for week in cfg.get("weeks", []):
        for mt in week.get("micro_topics", []):
            topics.append(MicroTopic(
                id=mt["id"],
                title=mt["title"],
                problem_num=mt.get("problem_num", 0),
                difficulty=mt.get("difficulty", 3),
                estimated_minutes=mt.get("estimated_minutes", 25),
                source_problems=mt.get("source_problems", []),
                key_techniques=mt.get("key_techniques", []),
                misconceptions=mt.get("misconceptions", []),
                generate=mt.get("generate", {"article": True, "problems_solved": 5, "practice_hints": 4}),
                week=week["num"],
                phase=week.get("phase", 1),
            ))
    return topics, cfg


# ═══════════════════════════════════════════════════════════
# OUTPUT MANAGER
# ═══════════════════════════════════════════════════════════
class PlanWriter:
    """Записва структурирано в plan_output/{topic_id}/."""
    
    def __init__(self):
        self.dir = PLAN_OUTPUT
    
    def topic_dir(self, topic_id: str) -> Path:
        d = self.dir / topic_id
        d.mkdir(exist_ok=True)
        return d
    
    def write_article(self, topic: MicroTopic):
        d = self.topic_dir(topic.id)
        art = topic.article
        md = f"""# {topic.title}

**ID:** `{topic.id}`
**Седмица:** {topic.week} | **Фаза:** {topic.phase}
**Problem #{topic.problem_num}** | Трудност: {topic.difficulty}/5 | ~{topic.estimated_minutes} мин

---

## Резюме

{art.get("summary", "")}

---

## Пълна статия

{art.get("deep", "")}
"""
        (d / "article.md").write_text(md, encoding="utf-8")
        (d / "article.json").write_text(
            json.dumps(art, indent=2, ensure_ascii=False), encoding="utf-8"
        )
    
    def write_problems(self, topic: MicroTopic):
        d = self.topic_dir(topic.id)
        data = topic.problems
        (d / "problems.json").write_text(
            json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8"
        )
        
        # Markdown версия за принтиране
        md = f"# {topic.title} — Тренировъчни задачи\n\n"
        for p in data.get("problems", []):
            md += f"## Задача {p['num']} (трудност {p.get('difficulty','?')}, ~{p.get('estimated_minutes','?')} мин)\n\n"
            md += f"{p['problem']}\n\n"
            md += f"<details><summary>Решение</summary>\n\n{p['solution']}\n\n**Отговор:** {p.get('answer','')}\n\n</details>\n\n"
            md += "---\n\n"
        (d / "problems.md").write_text(md, encoding="utf-8")
    
    def write_practice(self, topic: MicroTopic):
        d = self.topic_dir(topic.id)
        data = topic.practice
        (d / "practice.json").write_text(
            json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8"
        )
        
        md = f"# {topic.title} — Задачи за самостоятелна работа\n\n"
        for p in data.get("practice", []):
            md += f"## Задача {p['num']} (~{p.get('estimated_minutes','?')} мин)\n\n"
            md += f"{p['problem']}\n\n"
            md += f"<details><summary>Подсказка 1</summary>{p.get('hint1','')}</details>\n\n"
            md += f"<details><summary>Подсказка 2</summary>{p.get('hint2','')}</details>\n\n"
            md += f"<details><summary>Подсказка 3</summary>{p.get('hint3','')}</details>\n\n"
            md += f"<details><summary>Отговор</summary>{p.get('answer','')}</details>\n\n"
            md += "---\n\n"
        (d / "practice.md").write_text(md, encoding="utf-8")
    
    def save_meta(self, topic: MicroTopic):
        d = self.topic_dir(topic.id)
        meta = {
            "id": topic.id,
            "title": topic.title,
            "week": topic.week,
            "phase": topic.phase,
            "problem_num": topic.problem_num,
            "status": topic.status,
            "generated_at": time.time(),
        }
        (d / "meta.json").write_text(json.dumps(meta, indent=2), encoding="utf-8")


# ═══════════════════════════════════════════════════════════
# WORKER
# ═══════════════════════════════════════════════════════════
class PlanWorker:
    def __init__(self, week_filter: Optional[int] = None, topic_filter: Optional[str] = None):
        self.state = State()  # Reuse state from pipeline
        self.writer = PlanWriter()
        self.generator = AIClient(CONFIG["generator"], name="plan-gen")
        self.verifier = (
            AIClient(CONFIG["verifier"], name="plan-ver")
            if CONFIG.get("enable_verification") and CONFIG["verifier"].get("api_key")
            else None
        )
        
        self.topics, self.cfg = load_plan()
        if week_filter:
            self.topics = [t for t in self.topics if t.week == week_filter]
        if topic_filter:
            self.topics = [t for t in self.topics if t.id == topic_filter]
        
        self.week_filter = week_filter
        self.topic_filter = topic_filter
        self.processed = 0
        self.errors = 0
    
    def _get_state_key(self, topic_id: str, stage: str) -> str:
        return f"plan:{topic_id}:{stage}"
    
    def is_stage_done(self, topic_id: str, stage: str) -> bool:
        return self.state.is_done(self._get_state_key(topic_id, stage))
    
    def mark_stage(self, topic_id: str, stage: str, cost: float = 0.0):
        self.state.mark(self._get_state_key(topic_id, stage), "done", cost)
    
    # ─── STAGE 1: ARTICLE ───
    def generate_article(self, topic: MicroTopic) -> bool:
        if self.is_stage_done(topic.id, "article"):
            # Зареди от диск
            f = self.writer.topic_dir(topic.id) / "article.json"
            if f.exists():
                topic.article = json.loads(f.read_text(encoding="utf-8"))
                return True
        
        user = ARTICLE_PROMPT.format(
            problem_num=topic.problem_num,
            problem_description="Алгебра/уравнение с параметър" if topic.problem_num == 1 else "...",
            topic=topic.title,
        ) + f"\n\n═══ ИЗТОЧНИЦИ ═══\n" + "\n".join(
            f"- {sp['year']}/{sp['grade']} зад {sp['task']}: {sp.get('summary','')}"
            for sp in topic.source_problems
        )
        
        log.info(f"▶ [{topic.id}] Article")
        text, tokens = self.generator.call(ARTICLE_PROMPT, user, topic.id, force_json=True)
        if not text:
            log.error(f"✗ [{topic.id}] Article generation failed")
            return False
        
        parsed = _parse_json_safe(text)
        if not parsed:
            log.error(f"✗ [{topic.id}] Article JSON parse failed")
            return False
        
        topic.article = parsed
        self.writer.write_article(topic)
        self.mark_stage(topic.id, "article", 0.005)
        return True
    
    # ─── STAGE 2: SOLVED PROBLEMS ───
    def generate_problems(self, topic: MicroTopic) -> bool:
        if self.is_stage_done(topic.id, "problems"):
            f = self.writer.topic_dir(topic.id) / "problems.json"
            if f.exists():
                topic.problems = json.loads(f.read_text(encoding="utf-8"))
                return True
        
        n = topic.generate.get("problems_solved", 5)
        user = PROBLEMS_PROMPT.format(
            N=n, title=topic.title,
            key_techniques="\n".join(f"- {t}" for t in topic.key_techniques),
            misconceptions="\n".join(f"- {m}" for m in topic.misconceptions),
            source_problems="\n".join(
                f"- {sp['year']}/{sp['grade']} зад {sp['task']}: {sp.get('summary','')}"
                for sp in topic.source_problems
            ),
        )
        
        log.info(f"▶ [{topic.id}] Problems ({n})")
        text, tokens = self.generator.call(PROBLEMS_PROMPT, user, topic.id, force_json=True)
        if not text:
            return False
        
        parsed = _parse_json_safe(text)
        if not parsed or "problems" not in parsed:
            log.error(f"✗ [{topic.id}] Problems JSON invalid")
            return False
        
        # Verify count
        if len(parsed["problems"]) < n - 1:
            log.warning(f"⚠ [{topic.id}] Only {len(parsed['problems'])} problems (expected {n})")
        
        topic.problems = parsed
        self.writer.write_problems(topic)
        self.mark_stage(topic.id, "problems", 0.008)
        return True
    
    # ─── STAGE 3: PRACTICE ───
    def generate_practice(self, topic: MicroTopic) -> bool:
        if self.is_stage_done(topic.id, "practice"):
            f = self.writer.topic_dir(topic.id) / "practice.json"
            if f.exists():
                topic.practice = json.loads(f.read_text(encoding="utf-8"))
                return True
        
        n = topic.generate.get("practice_hints", 4)
        user = PRACTICE_PROMPT.format(
            N=n, title=topic.title,
            key_techniques="\n".join(f"- {t}" for t in topic.key_techniques),
        )
        
        log.info(f"▶ [{topic.id}] Practice ({n})")
        text, tokens = self.generator.call(PRACTICE_PROMPT, user, topic.id, force_json=True)
        if not text:
            return False
        
        parsed = _parse_json_safe(text)
        if not parsed or "practice" not in parsed:
            return False
        
        topic.practice = parsed
        self.writer.write_practice(topic)
        self.mark_stage(topic.id, "practice", 0.005)
        return True
    
    # ─── FULL PIPELINE FOR ONE TOPIC ───
    def process_topic(self, topic: MicroTopic) -> bool:
        if self.state.budget_exceeded():
            log.warning("Budget exceeded")
            return False
        
        log.info(f"═══ [{topic.id}] {topic.title} ═══")
        topic.attempts += 1
        
        try:
            if topic.generate.get("article"):
                if not self.generate_article(topic):
                    topic.status = "error"
                    topic.error = "article failed"
                    return False
            
            if topic.generate.get("problems_solved", 0) > 0:
                if not self.generate_problems(topic):
                    topic.status = "error"
                    topic.error = "problems failed"
                    return False
            
            if topic.generate.get("practice_hints", 0) > 0:
                if not self.generate_practice(topic):
                    topic.status = "error"
                    topic.error = "practice failed"
                    return False
            
            topic.status = "done"
            self.writer.save_meta(topic)
            self.processed += 1
            log.info(f"✓ [{topic.id}] DONE")
            return True
        
        except Exception as e:
            log.exception(f"✗ [{topic.id}] {e}")
            topic.status = "error"
            topic.error = str(e)
            self.errors += 1
            return False
    
    # ─── RUN MODES ───
    def run_once(self) -> int:
        pending = [t for t in self.topics if t.status != "done"]
        if not pending:
            return 0
        
        batch = pending[:CONFIG["batch_size"]]
        n = 0
        with ThreadPoolExecutor(max_workers=CONFIG["max_concurrent"]) as pool:
            futures = {pool.submit(self.process_topic, t): t for t in batch}
            for fut in as_completed(futures):
                if fut.result():
                    n += 1
        return n
    
    def run_forever(self):
        log.info("═" * 60)
        log.info("Plan Worker 24/7")
        log.info(f"  Topics: {len(self.topics)}")
        log.info(f"  Week filter: {self.week_filter or 'all'}")
        log.info(f"  Budget/day: ${CONFIG['daily_budget_usd']:.2f}")
        log.info("═" * 60)
        
        import signal
        def stop(*a): self.running = False
        self.running = True
        signal.signal(signal.SIGINT, stop)
        signal.signal(signal.SIGTERM, stop)
        
        while self.running:
            n = self.run_once()
            if n == 0 and all(t.status == "done" for t in self.topics):
                log.info("All topics done. Stopping.")
                break
            if self.state.budget_exceeded():
                log.warning("Budget 80% — pause 1h")
                time.sleep(3600)
                continue
            time.sleep(CONFIG["tick_interval_sec"])
    
    def print_stats(self):
        total = len(self.topics)
        done = sum(1 for t in self.topics if t.status == "done")
        err = sum(1 for t in self.topics if t.status == "error")
        print("═" * 55)
        print(f"  Total topics:  {total}")
        print(f"  ✓ Done:        {done} ({done/total*100:.1f}%)")
        print(f"  ✗ Error:       {err}")
        print(f"  Budget today:  ${self.state.data['budget']['spent_usd']:.3f}")
        print("═" * 55)


def main():
    import argparse
    ap = argparse.ArgumentParser()
    ap.add_argument("--once", action="store_true")
    ap.add_argument("--week", type=int)
    ap.add_argument("--topic")
    ap.add_argument("--stats", action="store_true")
    args = ap.parse_args()
    
    w = PlanWorker(week_filter=args.week, topic_filter=args.topic)
    if args.stats:
        w.print_stats()
    elif args.once:
        n = w.run_once()
        print(f"Processed: {n}")
        w.print_stats()
    else:
        w.run_forever()


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
plan_integrate.py — вгражда генерираното в index.html (нова PLAN структура).
Създава 2 неща:
  1. Embedded content в <script id="PLAN_DATA">
  2. Print-friendly view за всяка седмица
"""
import json
import re
from pathlib import Path

BASE_DIR = Path(__file__).parent.resolve()
PLAN_OUTPUT = BASE_DIR / "plan_output"


def load_all_topics() -> dict:
    """Чете всички topic директории."""
    result = {}
    for d in sorted(PLAN_OUTPUT.iterdir()):
        if not d.is_dir():
            continue
        meta_f = d / "meta.json"
        if not meta_f.exists():
            continue
        meta = json.loads(meta_f.read_text(encoding="utf-8"))
        
        topic = {
            "id": meta["id"],
            "title": meta["title"],
            "week": meta["week"],
            "phase": meta["phase"],
            "problem_num": meta["problem_num"],
            "article": None,
            "problems": None,
            "practice": None,
        }
        
        if (d / "article.json").exists():
            topic["article"] = json.loads((d / "article.json").read_text(encoding="utf-8"))
        if (d / "problems.json").exists():
            topic["problems"] = json.loads((d / "problems.json").read_text(encoding="utf-8"))
        if (d / "practice.json").exists():
            topic["practice"] = json.loads((d / "practice.json").read_text(encoding="utf-8"))
        
        result[topic["id"]] = topic
    return result


def build_plan_data_js(topics: dict, plan_config: dict) -> str:
    """Генерира JS с всички данни за вграждане."""
    # Групирай по седмица
    by_week = {}
    for tid, t in topics.items():
        by_week.setdefault(t["week"], []).append(t)
    
    weeks_data = []
    for wnum, wcfg in enumerate(plan_config.get("weeks", []), 1):
        if wcfg["num"] != wnum:
            wnum = wcfg["num"]
        week_topics = by_week.get(wnum, [])
        weeks_data.append({
            "num": wnum,
            "phase": wcfg.get("phase", 1),
            "main": wcfg.get("main", ""),
            "sec": wcfg.get("sec", ""),
            "topics": week_topics,
        })
    
    return (
        "const PLAN_DATA = "
        + json.dumps({"weeks": weeks_data}, ensure_ascii=False, indent=2)
        + ";\n"
    )


def embed_plan_data(html_path: Path, plan_data_js: str) -> str:
    """Вгражда PLAN_DATA в HTML."""
    html = html_path.read_text(encoding="utf-8")
    
    # Намери или създай <script id="PLAN_DATA">
    existing = re.search(
        r'<script\s+id="PLAN_DATA"[^>]*>.*?</script>',
        html, flags=re.DOTALL
    )
    
    new_script = f'<script id="PLAN_DATA">\n{plan_data_js}</script>'
    
    if existing:
        html = html[:existing.start()] + new_script + html[existing.end():]
    else:
        # Вмъкни преди </body>
        idx = html.rfind("</body>")
        if idx == -1:
            idx = len(html)
        html = html[:idx] + new_script + "\n" + html[idx:]
    
    return html

openTaskProblem(taskNum, problemId, year, summary) {
  const task = TASKS_DATA[taskNum];
  if (!task) return;
  // Try to find the source problem in STEPS
  const p = getPoint(problemId);
  if (p) {
    if (Tabs.current !== 'study') Tabs.switch('study');
    Modal.open(problemId);
  } else {
    Utils.toast(`Задача ${year}: отвори от архива (PDF)`, 'file-text');
  }
},

toggleTaskProblem(taskNum, problemId) {
  if (!Store.state.taskProgress) Store.state.taskProgress = {};
  const key = `T${taskNum}:${problemId}`;
  const prev = !!Store.state.taskProgress[key];
  Store.state.taskProgress[key] = !prev;
  History.push('toggleTask', { key, prev });
  Store.save();
  Content.render();
}

openTaskProblem(problemId, year, grade) {
  // First: try to find as STEPS point

    Modal.open(problemId);
    return;
  }
  // Fallback: build PDF URL
  const pdf = `Задачи_Минали_Години/Областен/OLI-2-Областен кръг-${year}-${grade} кл-probs`;
  window.open(Utils.pdfUrl(pdf, 1), '_blank');
},

toggleTaskProblem(taskNum, problemId) {
  if (!Store.state.taskProgress) Store.state.taskProgress = {};
  const key = `T${taskNum}:${problemId}`;
  const prev = !!Store.state.taskProgress[key];
  Store.state.taskProgress[key] = !prev;
  History.push('toggleTask', { key, prev });
  Store.save();
  if (Tabs.current === 'tasks') this.renderTasksView();
},

python
#!/usr/bin/env python3
"""Регенерира project_scan.txt във формат: '<size> KB  |  <path>'."""
import os
from pathlib import Path

OUTPUT = 'project_scan.txt'
SKIP_DIRS = {'.git', 'node_modules', '__pycache__', '.venv', 'venv', '.idea', '.vscode'}
SKIP_PREFIXES = ('.',)  # всички скрити папки/файлове

def fmt_size(n_bytes: int) -> str:
    kb = n_bytes / 1024.0
    if n_bytes % 1024 == 0:
        return str(n_bytes // 1024)
    return f"{kb:.1f}"

def main():
    # 1. Truncate output first — така при повторно пускане
    #    output файлът се вижда с размер 0 (както в примера)
    Path(OUTPUT).write_text('', encoding='utf-8')

    rows = []
    for dirpath, dirnames, filenames in os.walk('.'):
        # Премахни skip-натите директории in-place (os.walk ги пропуска)
        dirnames[:] = sorted(d for d in dirnames if d not in SKIP_DIRS)

        for fname in filenames:
            fpath = Path(dirpath) / fname
            try:
                size = fpath.stat().st_size
            except OSError:
                continue

            rel = str(fpath)
            # Махни './' или '.\' от началото
            if rel.startswith('./'):
                rel = rel[2:]
            elif rel.startswith('.\\'):
                rel = rel[2:]

            rows.append((size, f"{fmt_size(size)} KB  |  {rel}"))

    # Сортирай: първо по размер, после по път (за детерминистичност при равни)
    rows.sort(key=lambda x: (x[0], x[1].lower()))

    Path(OUTPUT).write_text(
        '\n'.join(line for _, line in rows) + '\n',
        encoding='utf-8',
    )

    total_size = sum(s for s, _ in rows) / 1024
    print(f"✓ {OUTPUT}: {len(rows)} файла, {total_size:.1f} KB общо")

if __name__ == '__main__':
    main()
